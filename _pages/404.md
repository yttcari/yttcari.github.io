(function () {
  var wrap = document.getElementById('teapot-wrap');
  if (!wrap) return;
  var x = 0, y = 0, tx = 0, ty = 0;      // current / target offset from home
  var RADIUS = 180;                       // how close the cursor can get
  var stage = wrap.parentNode;

  function onMove(e) {
    var r = wrap.getBoundingClientRect();
    var cx = r.left + r.width / 2, cy = r.top + r.height / 2;
    var dx = cx - e.clientX, dy = cy - e.clientY;
    var d = Math.sqrt(dx * dx + dy * dy) || 1;
    if (d < RADIUS) {
      var push = RADIUS - d;
      tx += (dx / d) * push;
      ty += (dy / d) * push;
      // keep it between the header and the footer
      var home = stage.getBoundingClientRect();
      var hx = home.left + home.width / 2, hy = home.top;
      var top = document.querySelector('.masthead').getBoundingClientRect().bottom;
      var bottom = document.querySelector('.page__footer').getBoundingClientRect().top;
      var minX = -hx + r.width / 2, maxX = window.innerWidth - hx - r.width / 2;
      var minY = top - hy, maxY = bottom - hy - r.height;
      tx = Math.max(minX, Math.min(maxX, tx));
      ty = Math.max(minY, Math.min(maxY, ty));
    }
  }

  function tick() {
    x += (tx - x) * 0.15;
    y += (ty - y) * 0.15;
    wrap.style.transform = 'translate(' + x + 'px,' + y + 'px)';
    requestAnimationFrame(tick);
  }

  // "Into the unknown": jump to a random http.cat status code
  var codes = [
    100,101,102,103,200,201,202,203,204,207,208,226,300,301,302,303,304,
    305,307,308,401,402,403,404,406,407,408,409,410,411,412,413,414,

    499,500,501,502,503,504,506,507,508,509,510,511,521,522,523,525,530,
    599
  ];
  var link = document.getElementById('unknown-link');
  if (link) link.addEventListener('click', function () {
    link.href = 'https://http.cat/status/' + codes[Math.floor(Math.random() * codes.length)];
  });

  window.addEventListener('pointermove', onMove);
  tick();
})();
</script>
